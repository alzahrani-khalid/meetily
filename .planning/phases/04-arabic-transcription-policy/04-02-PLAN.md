**Create `frontend/src/hooks/useWhisperDownloadState.ts`:**

```typescript
import { useState, useEffect, useCallback } from 'react';
import { listen } from '@tauri-apps/api/event';
import { WhisperAPI } from '@/lib/whisper';

interface WhisperDownloadState {
  isReady: boolean;
  isDownloading: boolean;
  progress: number;
  error: string | null;
}

export function useWhisperDownloadState(modelName: string): WhisperDownloadState & { startDownload: () => Promise<void>; retryDownload: () => Promise<void> } {
  const [isReady, setIsReady] = useState(false);
  const [isDownloading, setIsDownloading] = useState(false);
  const [progress, setProgress] = useState(0);
  const [error, setError] = useState<string | null>(null);

  // Check initial model state on mount (survives page reload per Pitfall 3)
  useEffect(() => {
    WhisperAPI.getAvailableModels().then(models => {
      const model = models.find((m: any) => m.name === modelName);
      if (model) {
        if (model.status === 'Available' || model.status === 'available') {
          setIsReady(true);
        } else if (typeof model.status === 'object' && model.status !== null && 'Downloading' in model.status) {
          setIsDownloading(true);
          setProgress(model.status.Downloading);
        }
      }
    }).catch(err => console.error('[useWhisperDownloadState] Failed to check models:', err));
  }, [modelName]);

  // Listen for download events
  useEffect(() => {
    const unlistenProgress = listen<any>('model-download-progress', (event) => {
      if (event.payload.modelName === modelName) {
        setIsDownloading(true);
        setError(null);
        setProgress(Math.round(event.payload.progress));
      }
    });

    const unlistenComplete = listen<any>('model-download-complete', (event) => {
      if (event.payload.modelName === modelName) {
        setIsReady(true);
        setIsDownloading(false);
        setProgress(100);
        setError(null);
      }
    });

    const unlistenError = listen<any>('model-download-error', (event) => {
      if (event.payload.modelName === modelName) {
        setIsDownloading(false);
        setError(event.payload.error || 'Download failed');
      }
    });

    return () => {
      unlistenProgress.then(fn => fn());
      unlistenComplete.then(fn => fn());
      unlistenError.then(fn => fn());
    };
  }, [modelName]);

  const startDownload = useCallback(async () => {
    try {
      setIsDownloading(true);
      setError(null);
      setProgress(0);
      await WhisperAPI.downloadModel(modelName);
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Download failed');
      setIsDownloading(false);
    }
  }, [modelName]);

  const retryDownload = useCallback(async () => {
    setError(null);
    await startDownload();
  }, [startDownload]);

  return { isReady, isDownloading, progress, error, startDownload, retryDownload };
}
```

**Create `frontend/src/components/WhisperDownloadGate.tsx`:**

A shared component per D-04 and UI-SPEC Component Inventory section 2. Uses the `useWhisperDownloadState` hook. Renders a disabled record button with inline progress when `large-v3` is not ready.

```tsx
'use client';

import { useEffect } from 'react';
import { useTranslations } from 'next-intl';
import { Progress } from '@/components/ui/progress';
import { useWhisperDownloadState } from '@/hooks/useWhisperDownloadState';

interface WhisperDownloadGateProps {
  modelName: string;
  onReady?: () => void;
  children: React.ReactNode; // The normal record button
}

export function WhisperDownloadGate({ modelName, onReady, children }: WhisperDownloadGateProps) {
  const { isReady, isDownloading, progress, error, startDownload, retryDownload } = useWhisperDownloadState(modelName);
  const t = useTranslations('recording');

  // Auto-start download if model not ready and not downloading
  useEffect(() => {
    if (!isReady && !isDownloading && !error) {
      startDownload();
    }
  }, [isReady, isDownloading, error, startDownload]);

  // Notify parent when ready
  useEffect(() => {
    if (isReady && onReady) {
      onReady();
    }
  }, [isReady, onReady]);

  // Model ready — render normal children (record button)
  if (isReady) {
    return <>{children}</>;
  }

  // Downloading state — disabled button + progress
  if (isDownloading) {
    return (
      <div className="flex flex-col items-center gap-2">
        <button
          disabled
          className="min-h-11 min-w-11 rounded-full bg-muted text-muted-foreground px-6 py-3 opacity-70 cursor-not-allowed"
          aria-disabled="true"
          aria-describedby="download-status"
        >
          {/* Placeholder for record icon in disabled state */}
        </button>
        <Progress value={progress} className="h-2 w-48" />
        <p id="download-status" className="text-sm text-muted-foreground text-center">
          {t('downloadingModel', { progress })}
        </p>
      </div>
    );
  }

  // Error state — retry affordance
  if (error) {
    return (
      <div className="flex flex-col items-center gap-2">
        <button
          disabled
          className="min-h-11 min-w-11 rounded-full bg-muted text-muted-foreground px-6 py-3 opacity-70 cursor-not-allowed"
          aria-disabled="true"
        >
          {/* Placeholder for record icon in disabled state */}
        </button>
        <button
          onClick={retryDownload}
          className="text-sm text-muted-foreground underline hover:text-foreground"
          role="button"
        >
          {t('modelDownloadError')}
        </button>
      </div>
    );
  }

  // Fallback — waiting to start
  return (
    <div className="flex flex-col items-center gap-2">
      <button
        disabled
        className="min-h-11 min-w-11 rounded-full bg-muted text-muted-foreground px-6 py-3 opacity-70 cursor-not-allowed"
        aria-disabled="true"
      >
        {/* Waiting state */}
      </button>
      <p className="text-sm text-muted-foreground text-center">
        {t('noArabicModel')}
      </p>
    </div>
  );
}
```

Both files use `text-sm`, `text-center` (safe -- not flipped by RTL), and no physical-direction classes. Touch target minimum 44x44 maintained via `min-h-11 min-w-11`.
